module if_register(

    input             clk,
    input             rst,

    input             Stall_D,
    input             Flush_D,

    input      [31:0] instruction_F,
    input      [31:0] PC_F,
    input      [31:0] PC_plus4_F,

    output reg [31:0] Instruction_D,
    output reg [31:0] PC_D,
    output reg [31:0] PCPlus4_D

);

always @(posedge clk) begin

    if(rst) begin

        Instruction_D <= 32'd0;
        PC_D          <= 32'd0;
        PCPlus4_D     <= 32'd0;

    end

    else if(Flush_D) begin

        // Insert Bubble (NOP)

        Instruction_D <= 32'd0;
        PC_D          <= 32'd0;
        PCPlus4_D     <= 32'd0;

    end

    else if(Stall_D) begin

        // Hold current values

        Instruction_D <= Instruction_D;
        PC_D          <= PC_D;
        PCPlus4_D     <= PCPlus4_D;

    end

    else begin

        Instruction_D <= instruction_F;
        PC_D          <= PC_F;
        PCPlus4_D     <= PC_plus4_F;

    end

end

endmodule
