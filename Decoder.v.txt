module Decoder(
  input  [6:0] opcode_in,
  input  [2:0] Func3_in,
  input  [6:0] Func7_in,

  output reg        Reg_wr_en_out,
  output reg [2:0]  imm_type_out,
  output reg        ladder_src_out,
  output reg        ALU_src_out,
  output reg [3:0]  ALU_control_out,
  output reg        DM_wr_en,
  output reg [7:0]  Branch_cond_out,
  output reg        Load_unsigned_out,
  output reg [1:0]  Load_size_out,
  output reg [2:0]  Result_src_out
);

  always @(*) begin

   Reg_wr_en_out     = 1'b0;
   imm_type_out      = 3'b000;
   ladder_src_out    = 1'b0;
   ALU_src_out       = 1'b0;
   ALU_control_out   = 4'b0000;
   DM_wr_en          = 1'b0;
   Branch_cond_out   = 8'b00000000;
   Load_unsigned_out = 1'b0;
   Load_size_out     = 2'b00;
   Result_src_out    = 3'b000;

    case(opcode_in)
      7'b0110011: begin

    Reg_wr_en_out  = 1'b1;
    ALU_src_out    = 1'b0;
    Result_src_out = 3'b000;

    case(Func3_in)

        3'b000: begin
            if(Func7_in == 7'b0100000)
                ALU_control_out = 4'b0001; // SUB
            else
                ALU_control_out = 4'b0000; // ADD
        end

        3'b111: ALU_control_out = 4'b0010; // AND
        3'b110: ALU_control_out = 4'b0011; // OR
        3'b100: ALU_control_out = 4'b0100; // XOR
        3'b001: ALU_control_out = 4'b0101; // SLL

        3'b101: begin
            if(Func7_in == 7'b0100000)
                ALU_control_out = 4'b0111; // SRA
            else
                ALU_control_out = 4'b0110; // SRL
        end

        3'b010: ALU_control_out = 4'b1000; // SLT
        3'b011: ALU_control_out = 4'b1001; // SLTU
      default:  ALU_control_out   = 4'b0000;
      
    endcase

end   

      //I- type instruction
      7'b0010011: begin

         Reg_wr_en_out  = 1'b1;
         imm_type_out   = 3'b000;
         ALU_src_out    = 1'b1;
         Result_src_out = 3'b000;

        case(Func3_in)

            3'b000: ALU_control_out = 4'b0000; //addi
            3'b111: ALU_control_out = 4'b0010; //andi
            3'b100: ALU_control_out = 4'b0100; //XORi
            3'b110: ALU_control_out = 4'b0011; //ORI
             default: ALU_control_out = 4'b0000;

        endcase

      end

      //load instruction
      7'b0000011: begin

      Reg_wr_en_out     = 1'b1;
      ALU_src_out       = 1'b1;
      Load_size_out     = 2'b00;
      Load_unsigned_out = 1'b0;
      Result_src_out    = 3'b001;
      //imm_type_out      = 3'b000;

        case(Func3_in)

                3'b000: begin
                    Load_size_out     = 2'b00; // LB
                    Load_unsigned_out = 1'b0;
                end

                3'b001: begin
                    Load_size_out     = 2'b01; // LH
                    Load_unsigned_out = 1'b0;
                end

                3'b010: begin
                    Load_size_out     = 2'b10; // LW
                    Load_unsigned_out = 1'b0;
                end

                3'b100: begin
                    Load_size_out     = 2'b00; // LBU
                    Load_unsigned_out = 1'b1;
                end

                3'b101: begin
                    Load_size_out     = 2'b01; // LHU
                    Load_unsigned_out = 1'b1;
                end
          default: begin
                  Load_size_out     = 2'b00;
                  Load_unsigned_out = 1'b0;
              end

        endcase

      end

      //store instruction
      7'b0100011: begin

        ALU_src_out     = 1'b1;
        imm_type_out    = 3'b001;
        ALU_control_out = 4'b0000;
        DM_wr_en        = 1'b1;

      end

      //branch instrction
      7'b1100011: begin

          Reg_wr_en_out  = 1'b0;
          ALU_src_out    = 1'b0;
          imm_type_out   = 3'b010;
          ladder_src_out = 1'b0;

          case(Func3_in)

            3'b000: Branch_cond_out = 8'b00000001; //BEQ
            3'b001: Branch_cond_out = 8'b00000010; //BNE
            3'b100: Branch_cond_out = 8'b00000100; //BLT
            3'b101: Branch_cond_out = 8'b00001000; //BGE
            default: Branch_cond_out = 8'b00000000;

          endcase

      end

      //jump Instruction

      //JAL instruction
      7'b1101111: begin

        Reg_wr_en_out  = 1'b1;
        imm_type_out   = 3'b100;
        Result_src_out = 3'b010;
        ladder_src_out = 1'b0;
        Branch_cond_out   = 8'b00010000; //JAL

      end

      //JALR
      7'b1100111: begin

        Reg_wr_en_out  = 1'b1;
        ALU_src_out    = 1'b1;
        ladder_src_out = 1'b1;
        imm_type_out   = 3'b000;
        Result_src_out = 3'b010;
        Branch_cond_out   = 8'b00100000;   // JALR

      end

      //LUI
      7'b0110111: begin

        Reg_wr_en_out  = 1'b1;
        imm_type_out   = 3'b011;
        Result_src_out = 3'b011;

      end

      //AUIPC
      7'b0010111: begin

        Reg_wr_en_out   = 1'b1;
        imm_type_out    = 3'b011;
        ALU_src_out     = 1'b1;
        ALU_control_out = 4'b0000;
        Result_src_out  = 3'b000;

      end
      default: begin
   
end

    endcase

  end

endmodule
