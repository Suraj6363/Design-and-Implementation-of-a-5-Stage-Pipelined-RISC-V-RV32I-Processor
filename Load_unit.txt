module load_unit(read_data_in,load_size_in,load_unsigned_in,loaded_data_out);
  input [31:0] read_data_in;
  input [1:0] load_size_in;
  input load_unsigned_in;
  output reg [31:0] loaded_data_out;
  
  always @(*) begin
    case(load_size_in)
      2'b00:begin
        if(load_unsigned_in)
          loaded_data_out={24'b0,read_data_in[7:0]};
        else
          loaded_data_out={{24{read_data_in[7]}},read_data_in[7:0]};
      end
      
      2'b01:begin
         if(load_unsigned_in)
           loaded_data_out={16'b0,read_data_in[15:0]};
        else
          loaded_data_out={{16{read_data_in[15]}},read_data_in[15:0]};
      end
      
      2'b10:begin
        loaded_data_out=read_data_in;
      end
      
      default:loaded_data_out=32'b0;
    endcase
  end
endmodule
