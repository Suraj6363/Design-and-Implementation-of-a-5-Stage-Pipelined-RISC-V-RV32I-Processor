`timescale 1ns/1ps

module tb;


//=============================================================
// CLOCK & RESET
//=============================================================

reg clk;
reg rst;

initial begin
    clk = 1'b0;
    forever #5 clk = ~clk;
end

initial begin
    rst = 1'b1;
    #20;
    rst = 1'b0;
end


//=============================================================
// DUT CONNECTIONS
//=============================================================

wire        DM_WrEn_out;
wire [3:0]  DM_mask_out;
wire [31:0] DM_addr_out;
wire [31:0] DM_data_out;
wire [31:0] instr_addr_out;

reg  [31:0] instruction_in;
reg  [31:0] DM_data_in;


//=============================================================
// DUT
//=============================================================

top_module dut
(
    .clk_in         (clk),
    .rst_in         (rst),

    .instruction_in (instruction_in),
    .DM_data_in     (DM_data_in),

    .instr_addr_out (instr_addr_out),

    .DM_WrEn_out    (DM_WrEn_out),
    .DM_mask_out    (DM_mask_out),
    .DM_addr_out    (DM_addr_out),
    .DM_data_out    (DM_data_out)
);


//=============================================================
// INSTRUCTION MEMORY
//=============================================================

reg [31:0] imem [0:255];

integer i;

initial begin

    //=========================================================
    // INITIAL REGISTER VALUES
    //=========================================================

    // ADDI x1, x0, 10
    imem[0] = 32'h00A00093;

    // ADDI x2, x0, 20
    imem[1] = 32'h01400113;


    //=========================================================
    // R-TYPE INSTRUCTIONS
    //=========================================================

    // ADD x3, x1, x2
    imem[2] = 32'h002081B3;

    // SUB x4, x1, x2
    imem[3] = 32'h40208233;

    // AND x5, x1, x2
    imem[4] = 32'h0020F2B3;

    // OR x6, x1, x2
    imem[5] = 32'h0020E333;

    // XOR x7, x1, x2
    imem[6] = 32'h0020C3B3;

    // SLL x8, x1, x2
    imem[7] = 32'h00209433;

    // SRL x9, x1, x2
    imem[8] = 32'h0020D4B3;

    // SRA x10, x1, x2
    imem[9] = 32'h4020D533;

    // SLT x11, x1, x2
    imem[10] = 32'h0020A5B3;

    // SLTU x12, x1, x2
    imem[11] = 32'h0020B633;


    //=========================================================
    // I-TYPE ALU INSTRUCTIONS
    //=========================================================

    // ADDI x13, x1, 5
    imem[12] = 32'h00508693;

    // ANDI x14, x1, 15
    imem[13] = 32'h00F0F713;

    // XORI x15, x1, 5
    imem[14] = 32'h0050C793;

    // ORI x16, x1, 3
    imem[15] = 32'h0030E813;


    //=========================================================
    // LOAD INSTRUCTIONS
    //=========================================================

    // LB x17, 0(x0)
    imem[16] = 32'h00000883;

    // LH x18, 0(x0)
    imem[17] = 32'h00001903;

    // LW x19, 0(x0)
    imem[18] = 32'h00002983;

    // LBU x20, 0(x0)
    imem[19] = 32'h00004A03;

    // LHU x21, 0(x0)
    imem[20] = 32'h00005A83;


    //=========================================================
    // STORE INSTRUCTIONS
    //=========================================================

    // SB x3, 4(x0)
    imem[21] = 32'h00300223;

    // SH x3, 8(x0)
    imem[22] = 32'h00301423;

    // SW x3, 12(x0)
    imem[23] = 32'h00302623;


    //=========================================================
    // LOAD-USE HAZARD TEST
    //=========================================================

    // LW x22, 12(x0)
    imem[24] = 32'h00C02B03;

    // ADD x23, x22, x1
    //
    // x22 is produced by the immediately preceding LW.
    // This instruction is intentionally placed here to test
    // the load-use hazard / stall logic.
    imem[25] = 32'h001B0BB3;


    //=========================================================
    // BRANCH TESTS
    //=========================================================

    // BEQ x1, x2, +8
    // x1=10, x2=20 -> NOT TAKEN
    imem[26] = 32'h00208463;

    // BNE x1, x2, +8
    // x1=10, x2=20 -> TAKEN
    // Jumps from instruction 27 to instruction 29.
    imem[27] = 32'h00209463;

    // This instruction should be skipped by BNE
    // ADDI x28, x0, 111
    imem[28] = 32'h06F00E13;

    // BLT x1, x2, +8
    // x1=10, x2=20 -> TAKEN
    // Jumps from instruction 29 to instruction 31.
    imem[29] = 32'h0020C463;

    // This instruction should be skipped by BLT
    // ADDI x28, x0, 222
    imem[30] = 32'h0DE00E13;

    // BGE x1, x2, +8
    // x1=10, x2=20 -> NOT TAKEN
    imem[31] = 32'h0020D463;


    //=========================================================
    // JAL TEST
    //=========================================================

    // JAL x22, +8
    //
    // Current instruction index = 32
    // Target index = 34
    //
    // x22 receives PC+4.
    imem[32] = 32'h00800B6F;

    // This instruction should be skipped by JAL
    // ADDI x28, x0, 333
    imem[33] = 32'h14D00E13;


    //=========================================================
    // JALR PREPARATION
    //=========================================================

    // ADDI x23, x0, 148
    //
    // 148 decimal = 0x94
    // Target instruction address = 0x94
    // Instruction index = 37
    imem[34] = 32'h09400B93;


    //=========================================================
    // JALR TEST
    //=========================================================

    // JALR x24, x23, 0
    //
    // Target = x23 + 0 = 148
    // Therefore execution jumps to instruction index 37.
    //
    // x24 receives PC+4.
    imem[35] = 32'h000B8C67;

    // This instruction should be skipped by JALR
    // ADDI x28, x0, 444
    imem[36] = 32'h1BC00E13;


    //=========================================================
    // JALR TARGET
    //=========================================================

    // LUI x25, 0x12345
    imem[37] = 32'h12345CB7;

    // AUIPC x26, 0x1
    imem[38] = 32'h00001D17;

    // NOP
    imem[39] = 32'h00000013;


    //=========================================================
    // FILL REMAINING MEMORY WITH NOP
    //=========================================================

    for(i = 40; i < 256; i = i + 1)
        imem[i] = 32'h00000013;

end


//=============================================================
// INSTRUCTION MEMORY READ
//=============================================================

always @(*)
begin
    instruction_in = imem[instr_addr_out[31:2]];
end


//=============================================================
// DATA MEMORY
//=============================================================

reg [31:0] dmem [0:255];

initial begin

    for(i = 0; i < 256; i = i + 1)
        dmem[i] = 32'd0;

    // Special value for testing:
    //
    // Byte  = 80 -> LB  should sign extend
    // Byte  = 80 -> LBU should zero extend
    //
    // Half = 7F80 -> LH/LHU can be checked
    //
    dmem[0] = 32'h80FF7F80;

end


//=============================================================
// DATA MEMORY READ
//=============================================================

always @(*)
begin
    DM_data_in = dmem[DM_addr_out[31:2]];
end


//=============================================================
// DATA MEMORY WRITE
//=============================================================

always @(posedge clk)
begin

    if(DM_WrEn_out)
    begin

        case(DM_mask_out)

            // WORD STORE
            4'b1111:
                dmem[DM_addr_out[31:2]]
                    <= DM_data_out;

            // HALF-WORD STORE
            4'b0011:
                dmem[DM_addr_out[31:2]][15:0]
                    <= DM_data_out[15:0];

            // BYTE STORE
            4'b0001:
                dmem[DM_addr_out[31:2]][7:0]
                    <= DM_data_out[7:0];

            default:
                dmem[DM_addr_out[31:2]]
                    <= dmem[DM_addr_out[31:2]];

        endcase

    end

end


//=============================================================
// PIPELINE MONITOR
//=============================================================

always @(posedge clk)
begin

    if(!rst)
    begin

        $display("");
        $display("------------------------------------------------------------");

        $display(
            "TIME=%0t  PC=%08h  INSTR=%08h",
            $time,
            instr_addr_out,
            instruction_in
        );

        // WRITE BACK
        $display(
            "WB    : RegWrite=%b  RD=%02d  Data=%08h",
            dut.RegWrite_W,
            dut.RD_W,
            dut.WB_Data
        );

        // MEMORY
        $display(
            "MEM   : RegWrite=%b  RD=%02d",
            dut.RegWrite_M,
            dut.RD_M
        );

        // EXECUTE
        $display(
            "EX    : RegWrite=%b  RD=%02d",
            dut.RegWrite_E,
            dut.RD_E
        );

        // FORWARDING
        $display(
            "FWD   : ForwardA=%b  ForwardB=%b",
            dut.ForwardA_E,
            dut.ForwardB_E
        );

        // STALL
        $display(
            "STALL : Stall_F=%b  Stall_D=%b",
            dut.Stall_F,
            dut.Stall_D
        );

        // FLUSH / BRANCH
        $display(
            "CTRL  : Flush_D=%b  Flush_E=%b  BranchTaken=%b",
            dut.Flush_D,
            dut.Flush_E,
            dut.BranchTaken_E
        );

        // DATA MEMORY
        $display(
            "DMEM  : WrEn=%b  Mask=%b  Addr=%08h  Data=%08h",
            DM_WrEn_out,
            DM_mask_out,
            DM_addr_out,
            DM_data_out
        );

    end

end


//=============================================================
// SELF-CHECKING INFRASTRUCTURE
//=============================================================

integer pass_count;
integer fail_count;

// Compares one register against its expected value.
// Prints PASS/FAIL for that register and updates counters.
// Fixed-width, left-justified columns so every row lines up:
//   [6-char PASS/FAIL] [4-char reg id] [28-char label] [got] [exp]
task check_reg;
    input [4:0]  reg_num;
    input [31:0] actual;
    input [31:0] expected;
    input [8*28:1] label; // description string, up to 28 chars
    reg    [8*8:1] reg_id;
    begin
        reg_id = "";
        $sformat(reg_id, "x%0d", reg_num);
        if (actual === expected) begin
            pass_count = pass_count + 1;
            $display("%-6s %-5s %-28s got=0x%08h  exp=0x%08h",
                      "PASS", reg_id, label, actual, expected);
        end
        else begin
            fail_count = fail_count + 1;
            $display("%-6s %-5s %-28s got=0x%08h  exp=0x%08h  <-- MISMATCH",
                      "FAIL", reg_id, label, actual, expected);
        end
    end
endtask

// Compares one data memory word against its expected value.
// Same fixed-width formatting as check_reg for alignment.
task check_mem;
    input [7:0]  word_idx;
    input [31:0] actual;
    input [31:0] expected;
    input [8*28:1] label;
    reg    [8*8:1] mem_id;
    begin
        mem_id = "";
        $sformat(mem_id, "MEM[%0d]", word_idx);
        if (actual === expected) begin
            pass_count = pass_count + 1;
            $display("%-6s %-5s %-28s got=0x%08h  exp=0x%08h",
                      "PASS", mem_id, label, actual, expected);
        end
        else begin
            fail_count = fail_count + 1;
            $display("%-6s %-5s %-28s got=0x%08h  exp=0x%08h  <-- MISMATCH",
                      "FAIL", mem_id, label, actual, expected);
        end
    end
endtask


//=============================================================
// FINAL RESULT
//=============================================================

initial
begin

    // More time because the test program is larger
    #1000;

    //=========================================================
    // FULL REGISTER FILE DUMP (x0 - x31)
    //=========================================================

    $display("");
    $display("============================================================");
    $display("                    FINAL REGISTER FILE");
    $display("============================================================");

    for (i = 0; i < 32; i = i + 1) begin
        $display("x%0d = %11d   (0x%08h)",
                  i,
                  $signed(dut.REGISTER_FILE.reg_file[i]),
                  dut.REGISTER_FILE.reg_file[i]);
    end


    //=========================================================
    // DATA MEMORY DUMP
    //=========================================================

    $display("");
    $display("============================================================");
    $display("                     DATA MEMORY");
    $display("============================================================");

    for (i = 0; i < 4; i = i + 1) begin
        $display("MEM[%0d]         = %11d   (0x%08h)",
                  i, dmem[i], dmem[i]);
    end


    //=========================================================
    // SELF-CHECKING: EXPECTED VS ACTUAL
    //=========================================================

    pass_count = 0;
    fail_count = 0;

    $display("");
    $display("============================================================");
    $display("                    SELF-CHECK RESULTS");
    $display("============================================================");

    // ---- R-type ----
    check_reg( 1, dut.REGISTER_FILE.reg_file[1],  32'd10,          "ADDI x1,x0,10");
    check_reg( 2, dut.REGISTER_FILE.reg_file[2],  32'd20,          "ADDI x2,x0,20");
    check_reg( 3, dut.REGISTER_FILE.reg_file[3],  32'd30,          "ADD  x3,x1,x2");
    check_reg( 4, dut.REGISTER_FILE.reg_file[4],  32'hFFFFFFF6,    "SUB  x4,x1,x2 (-10)");
    check_reg( 5, dut.REGISTER_FILE.reg_file[5],  32'd0,           "AND  x5,x1,x2");
    check_reg( 6, dut.REGISTER_FILE.reg_file[6],  32'd30,          "OR   x6,x1,x2");
    check_reg( 7, dut.REGISTER_FILE.reg_file[7],  32'd30,          "XOR  x7,x1,x2");
    check_reg( 8, dut.REGISTER_FILE.reg_file[8],  32'h00A00000,    "SLL  x8,x1,x2");
    check_reg( 9, dut.REGISTER_FILE.reg_file[9],  32'd0,           "SRL  x9,x1,x2");
    check_reg(10, dut.REGISTER_FILE.reg_file[10], 32'd0,           "SRA  x10,x1,x2");
    check_reg(11, dut.REGISTER_FILE.reg_file[11], 32'd1,           "SLT  x11,x1,x2");
    check_reg(12, dut.REGISTER_FILE.reg_file[12], 32'd1,           "SLTU x12,x1,x2");

    // ---- I-type ALU ----
    check_reg(13, dut.REGISTER_FILE.reg_file[13], 32'd15,          "ADDI x13,x1,5");
    check_reg(14, dut.REGISTER_FILE.reg_file[14], 32'd10,          "ANDI x14,x1,15");
    check_reg(15, dut.REGISTER_FILE.reg_file[15], 32'd15,          "XORI x15,x1,5");
    check_reg(16, dut.REGISTER_FILE.reg_file[16], 32'd11,          "ORI  x16,x1,3");

    // ---- Loads ----
    check_reg(17, dut.REGISTER_FILE.reg_file[17], 32'hFFFFFF80,    "LB   x17 (-128)");
    check_reg(18, dut.REGISTER_FILE.reg_file[18], 32'h00007F80,    "LH   x18");
    check_reg(19, dut.REGISTER_FILE.reg_file[19], 32'h80FF7F80,    "LW   x19");
    check_reg(20, dut.REGISTER_FILE.reg_file[20], 32'h00000080,    "LBU  x20");
    check_reg(21, dut.REGISTER_FILE.reg_file[21], 32'h00007F80,    "LHU  x21");

    // ---- Load-use hazard / jumps (final overwritten values) ----
    check_reg(22, dut.REGISTER_FILE.reg_file[22], 32'd132,         "x22 final (JAL PC+4)");
    check_reg(23, dut.REGISTER_FILE.reg_file[23], 32'd148,         "x23 final (ADDI 148)");
    check_reg(24, dut.REGISTER_FILE.reg_file[24], 32'd144,         "JALR x24 (PC+4)");

    // ---- Branch-skip guard: x28 must never be written ----
    check_reg(28, dut.REGISTER_FILE.reg_file[28], 32'd0,           "x28 must stay 0 (skipped)");

    // ---- LUI / AUIPC ----
    check_reg(25, dut.REGISTER_FILE.reg_file[25], 32'h12345000,    "LUI  x25");
    check_reg(26, dut.REGISTER_FILE.reg_file[26], 32'h00001098,    "AUIPC x26 (PC+imm)");

    // ---- x0 must always read 0 ----
    check_reg( 0, dut.REGISTER_FILE.reg_file[0],  32'd0,           "x0 hard-wired zero");

    // ---- Data memory ----
    check_mem(1, dmem[1], 32'd30, "SB x3,4(x0)");
    check_mem(2, dmem[2], 32'd30, "SH x3,8(x0)");
    check_mem(3, dmem[3], 32'd30, "SW x3,12(x0)");

    //=========================================================
    // SUMMARY
    //=========================================================

    $display("");
    $display("============================================================");
    $display("                    TEST SUMMARY");
    $display("============================================================");
    $display("PASSED = %0d", pass_count);
    $display("FAILED = %0d", fail_count);

    if (fail_count == 0)
        $display("RESULT : ALL TESTS PASSED");
    else
        $display("RESULT : %0d TEST(S) FAILED - see FAIL lines above", fail_count);

    $display("============================================================");


    //=========================================================
    // SIMULATION COMPLETE
    //=========================================================

    $display("");
    $display("============================================================");
    $display("                  SIMULATION COMPLETE");
    $display("============================================================");



    $finish;

end


endmodule
