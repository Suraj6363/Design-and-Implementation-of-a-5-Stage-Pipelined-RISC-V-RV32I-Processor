module register_file(
    input clk,
    input rst,
    input wr_en,
    input [4:0] des_addr_in,
    input [31:0] des_data_in,
    input [4:0] src_addr1_in,
    input [4:0] src_addr2_in,
    output [31:0] src_data1_out,
    output [31:0] src_data2_out
);

    reg [31:0] reg_file[31:0];
    integer i;

    always @(posedge clk or posedge rst) begin
        if(rst) begin
            for(i=0;i<32;i=i+1)
                reg_file[i] <= 32'd0;
        end
        else begin
            reg_file[0] <= 32'd0;
            if(wr_en && (des_addr_in != 5'd0))
                reg_file[des_addr_in] <= des_data_in;
        end
    end

    // Read-during-write bypass (write-through):
    // if the register being written this cycle is also being read,
    // forward the new data directly instead of the stale array value.
    assign src_data1_out =
        (wr_en && (des_addr_in != 5'd0) && (des_addr_in == src_addr1_in)) ?
        des_data_in : reg_file[src_addr1_in];

    assign src_data2_out =
        (wr_en && (des_addr_in != 5'd0) && (des_addr_in == src_addr2_in)) ?
        des_data_in : reg_file[src_addr2_in];

endmodule
