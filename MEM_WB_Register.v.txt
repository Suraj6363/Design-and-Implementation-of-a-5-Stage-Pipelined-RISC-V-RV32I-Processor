module MEM_WB_Register(

    input         clk,
    input         rst,

    // Control Signals

    input         RegWrite_M,
    input  [2:0]  ResultSrc_M,

    // Datapath Signals

    input  [31:0] ReadData_M,
    input  [31:0] ALUResult_M,
    input  [31:0] ImmExt_M,
    input  [31:0] PCPlus4_M,

    input  [4:0]  RD_M,

    // Control Outputs

    output reg        RegWrite_W,
    output reg [2:0]  ResultSrc_W,

    // Datapath Outputs

    output reg [31:0] ReadData_W,
    output reg [31:0] ALUResult_W,
    output reg [31:0] ImmExt_W,
    output reg [31:0] PCPlus4_W,

    output reg [4:0]  RD_W

);

always @(posedge clk) begin

    if(rst) begin

        RegWrite_W   <= 1'b0;
        ResultSrc_W  <= 3'b000;

        ReadData_W   <= 32'd0;
        ALUResult_W  <= 32'd0;
        ImmExt_W     <= 32'd0;
        PCPlus4_W    <= 32'd0;

        RD_W         <= 5'd0;

    end

    else begin

        RegWrite_W   <= RegWrite_M;
        ResultSrc_W  <= ResultSrc_M;

        ReadData_W   <= ReadData_M;
        ALUResult_W  <= ALUResult_M;
        ImmExt_W     <= ImmExt_M;
        PCPlus4_W    <= PCPlus4_M;

        RD_W         <= RD_M;

    end

end

endmodule
